- [ ] **Step 2: 添加初始化下拉刷新函数**

在 `<script>` 标签中添加以下函数：

```javascript
// 初始化下拉刷新
function initPullToRefresh() {
  const container = document.getElementById('app-root');
  const indicator = document.getElementById('pullToRefreshIndicator');
  
  // 触摸开始
  container.addEventListener('touchstart', (e) => {
    if (isRefreshing) return;
    startY = e.touches[0].clientY;
    isPulling = true;
  }, { passive: true });
  
  // 触摸移动
  container.addEventListener('touchmove', (e) => {
    if (!isPulling || isRefreshing) return;
    
    currentY = e.touches[0].clientY;
    const pullDistance = currentY - startY;
    
    // 只有在页面顶部且向下拉时才触发
    if (window.scrollY <= 0 && pullDistance > 0) {
      e.preventDefault(); // 阻止默认滚动
      
      const progress = Math.min(pullDistance / PULL_THRESHOLD, 1);
      
      indicator.classList.remove('hidden');
      indicator.style.transform = `translateY(${Math.min(pullDistance * 0.5, 40)}px)`;
      
      // 更新提示文本
      const text = indicator.querySelector('.refresh-text');
      const icon = indicator.querySelector('.refresh-icon');
      
      if (pullDistance > PULL_THRESHOLD) {
        text.textContent = '释放刷新...';
        icon.classList.remove('loading');
      } else {
        text.textContent = '下拉刷新...';
        icon.classList.remove('loading');
      }
    }
  }, { passive: false });
  
  // 触摸结束
  container.addEventListener('touchend', () => {
    if (!isPulling || isRefreshing) return;
    
    const pullDistance = currentY - startY;
    const indicator = document.getElementById('pullToRefreshIndicator');
    
    if (pullDistance > PULL_THRESHOLD) {
      // 触发刷新
      isRefreshing = true;
      indicator.style.transform = 'translateY(40px)';
      indicator.querySelector('.refresh-text').textContent = '刷新中...';
      indicator.querySelector('.refresh-icon').classList.add('loading');
      
      // 模拟刷新操作
      setTimeout(() => {
        refreshNews();
        hideRefreshIndicator();
        isRefreshing = false;
      }, 1500);
    } else {
      // 取消刷新
      hideRefreshIndicator();
    }
    
    isPulling = false;
    startY = 0;
    currentY = 0;
  });
}
```

```javascript
// 隐藏刷新指示器
function hideRefreshIndicator() {
  const indicator = document.getElementById('pullToRefreshIndicator');
  indicator.style.transform = 'translateY(-100%)';
  setTimeout(() => {
    indicator.classList.add('hidden');
  }, 300);
}
```

- [ ] **Step 4: 添加刷新新闻内容函数**

在 `<script>` 标签中添加以下函数：

```javascript
// 刷新新闻内容
function refreshNews() {
  const newsList = document.getElementById('newsList');
  const items = newsList.querySelectorAll('.relative');
  
  // 添加刷新动画
  items.forEach((item, index) => {
    item.classList.add('news-item-enter');
    item.style.animationDelay = `${index * 0.05}s`;
  });
  
  // 显示刷新成功提示
  showToast('新闻已刷新');
  
  // 移除动画类
  setTimeout(() => {
    items.forEach(item => {
      item.classList.remove('news-item-enter');
    });
  }, 500);
}
```
